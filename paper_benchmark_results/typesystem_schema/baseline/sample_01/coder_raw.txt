```python
import pytest
from typesystem.schemas import Schema
from typesystem.fields import Integer, String

def test_schema_default_application():
    # Field "a" has a default, so it is not required.
    schema = Schema(fields={"a": Integer(default=5), "b": String()})
    # Provide only the required field "b".
    input_data = {"b": "hello"}
    # Validation should succeed and apply the default for "a".
    validated = schema.validate(input_data)
    assert validated == {"a": 5, "b": "hello"}
```
